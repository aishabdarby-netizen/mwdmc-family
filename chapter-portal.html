async function loadChapterData() {

  if (!currentChapterId) return;

  try {

    const snap = await db.collection('chapters').doc(currentChapterId).get();

    if (snap.exists) {

      const d = snap.data();

      ['chapterName','location','repName','repContact','president','adults','children','dietary','tshirts','guests','news','message'].forEach(f => {

        const el = document.getElementById(f);

        if (el && d[f] !== undefined) el.value = d[f];

      });

      if (d._lastSaved) {

        const ts = new Date(d._lastSaved.toDate ? d._lastSaved.toDate() : d._lastSaved);

        ['contact','attendance','news'].forEach(s => {

          const el = document.getElementById(s + '-saved');

          if (el) el.textContent = 'Last saved: ' + ts.toLocaleString();

        });

      }

    }

  } catch(e) { showAlert('Could not load chapter data: ' + e.message, 'danger'); }

}


async function saveSection(section) {

  if (!currentChapterId) { showAlert('No chapter assigned.', 'danger'); return; }

  const fieldSets = {

    contact:    ['chapterName','location','repName','repContact','president'],

    attendance: ['adults','children','dietary','tshirts','guests'],

    news:       ['news','message']

  };

  const data = { _lastSaved: firebase.firestore.FieldValue.serverTimestamp() };

  fieldSets[section].forEach(f => { const el = document.getElementById(f); if (el) data[f] = el.value.trim(); });

  try {

    await db.collection('chapters').doc(currentChapterId).set(data, { merge: true });

    document.getElementById(section + '-saved').textContent = 'Saved at ' + new Date().toLocaleString() + ' ✓';

    showAlert('Chapter information saved successfully.', 'success');

    loadPublicView();

  } catch(e) { showAlert('Save failed: ' + e.message, 'danger'); }

}


async function loadPublicView() {

  const grid = document.getElementById('chapters-grid');

  if (!grid) return;

  grid.innerHTML = '<p style="color:var(--text-muted);font-size:0.9rem;font-family:Arial,sans-serif">Loading chapters...</p>';

  try {

    const snap = await db.collection('chapters').get();

    if (snap.empty) {

      grid.innerHTML = '<p style="color:var(--text-muted);font-size:0.9rem;font-family:Arial,sans-serif;grid-column:1/-1">No chapter information has been submitted yet.</p>';

      return;

    }

    grid.innerHTML = '';

    snap.forEach(doc => {

      const d = doc.data();

      const name = d.chapterName || CHAPTER_NAMES[doc.id] || doc.id;

      const card = document.createElement('div');

      card.className = 'chapter-card';

      card.innerHTML = `

        <h3>${esc(name)}</h3>

        <dl>

          ${d.location  ? `<dt>Location</dt><dd>${esc(d.location)}</dd>` : ''}

          ${d.repName   ? `<dt>Chapter Rep</dt><dd>${esc(d.repName)}</dd>` : ''}

          ${d.president ? `<dt>Leader</dt><dd>${esc(d.president)}</dd>` : ''}

          ${(d.adults||d.children) ? `<dt>Expected Attendance</dt><dd>${esc(d.adults||'?')} adults, ${esc(d.children||'?')} children</dd>` : ''}

          ${d.dietary   ? `<dt>Dietary / Access</dt><dd>${esc(d.dietary)}</dd>` : ''}

          ${d.tshirts   ? `<dt>T-Shirt Sizes</dt><dd>${esc(d.tshirts)}</dd>` : ''}

          ${d.guests    ? `<dt>Special Guests</dt><dd>${esc(d.guests)}</dd>` : ''}

          ${d.news      ? `<dt>Chapter News</dt><dd>${esc(d.news)}</dd>` : ''}

          ${d.message   ? `<dt>Message to the Family</dt><dd>${esc(d.message)}</dd>` : ''}

        </dl>`;

      grid.appendChild(card);

    });

  } catch(e) {

    grid.innerHTML = '<p style="color:var(--text-muted);font-size:0.9rem;font-family:Arial,sans-serif">Could not load chapter data. Please check your Firebase configuration.</p>';

  }

}

    else {

      let html = '<table style="width:100%;border-collapse:collapse">';

      html += '<tr style="background:var(--bg)"><th style="text-align:left;padding:6px 8px;border-bottom:1px solid var(--border)">Chapter</th><th style="text-align:left;padding:6px 8px;border-bottom:1px solid var(--border)">Rep</th><th style="text-align:left;padding:6px 8px;border-bottom:1px solid var(--border)">Attendance</th></tr>';

      snap.forEach(doc => {

        const d = doc.data();

        html += `<tr><td style="padding:6px 8px;border-bottom:0.5px solid var(--border)">${esc(d.chapterName||CHAPTER_NAMES[doc.id]||doc.id)}</td><td style="padding:6px 8px;border-bottom:0.5px solid var(--border)">${esc(d.repName||'—')}</td><td style="padding:6px 8px;border-bottom:0.5px solid var(--border)">${esc(d.adults||'?')} adults, ${esc(d.children||'?')} children</td></tr>`;

      });

      html += '</table>';

      adminList.innerHTML = html;

    }

    const entries = Object.entries(chapterMap);

    document.getElementById('admin-user-list').innerHTML = entries.length === 0

      ? '<div class="user-row">No assignments configured.</div>'

      : entries.map(([email, chap]) => `<div class="user-row"><span>${esc(email)}</span><span style="color:var(--primary);font-weight:bold">${esc(CHAPTER_NAMES[chap]||chap)}</span></div>`).join('');

  } catch(e) { document.getElementById('admin-data-list').innerHTML = '<p>Error: ' + e.message + '</p>'; }

}


function showAlert(msg, type) {

  const el = document.getElementById('alert-banner');

  el.textContent = msg;

  el.className = 'alert ' + type + ' show';

  setTimeout(() => { el.className = 'alert'; }, 5000);

}


function esc(s) {

  if (!s) return '';

  return String(s).replace(/&/g,'&amp;').replace(/</g,'&lt;').replace(/>/g,'&gt;').replace(/"/g,'&quot;');

}


function friendlyAuthError(code) {

    'auth/too-many-requests': 'Too many attempts. Please try again later.',

    'auth/invalid-credential': 'Email or password is incorrect.',

  };

  return map[code] || code;

}
